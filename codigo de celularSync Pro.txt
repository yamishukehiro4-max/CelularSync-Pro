function createStars(){
  const container=document.getElementById('starsBg');
  const layers=[
    {cls:'l1',count:120,min:.6,max:1.2,colors:['','']},
    {cls:'l2',count:80,min:1,max:1.8,colors:['','neon']},
    {cls:'l3',count:50,min:1.5,max:2.6,colors:['neon','warm','']}
  ];
  layers.forEach(layer=>{
    const div=document.createElement('div');
    div.className='star-layer '+layer.cls;
    for(let i=0;i<layer.count;i++){
      const s=document.createElement('span');
      s.className='star '+(layer.colors[i%layer.colors.length]||'');
      s.style.left=Math.random()*100+'%';
      s.style.top=Math.random()*100+'%';
      const size=layer.min+Math.random()*(layer.max-layer.min);
      s.style.width=size+'px';
      s.style.height=size+'px';
      s.style.animationDelay=(Math.random()*4)+'s';
      s.style.animationDuration=(2.5+Math.random()*3)+'s';
      div.appendChild(s);
    }
    container.appendChild(div);
  });
  setInterval(()=>{
    const shoot=document.createElement('span');
    shoot.className='star shooting';
    shoot.style.left=(Math.random()*40)+'%';
    shoot.style.top=(Math.random()*30)+'%';
    container.appendChild(shoot);
    setTimeout(()=>shoot.remove(),3200);
  },4500);
}
window.addEventListener('scroll',()=>{
  const y=window.pageYOffset;
  document.querySelectorAll('.star-layer').forEach((layer,i)=>{
    const speed=0.03+(i*0.04);
    layer.style.transform=`translateY(${y*speed}px)`;
  });
});

async function hashPassword(password){
  if(window.crypto&&window.crypto.subtle){
    const data=new TextEncoder().encode(password);
    const digest=await window.crypto.subtle.digest('SHA-256',data);
    return Array.from(new Uint8Array(digest)).map(b=>b.toString(16).padStart(2,'0')).join('');
  }
  let hash=2166136261;
  for(let i=0;i<password.length;i++){hash^=password.charCodeAt(i);hash=Math.imul(hash,16777619);}
  return 'fallback-'+(hash>>>0).toString(16);
}

let bubbleRotation=0;
let bubbleIntervalId=null;
const BUBBLE_COUNT=8;

function initHeroBubbles(){
  const container=document.getElementById('bubbleSystem');
  if(!container)return;
  // Limpiar burbujas previas
  container.querySelectorAll('.small-bubble').forEach(b=>b.remove());

  // Tomar los primeros 8 productos
  const heroProducts=products.slice(0, BUBBLE_COUNT);

  // Burbuja grande inicial con el primer producto
  const bigImg=document.getElementById('bigBubbleImg');
  if(heroProducts[0]){
    bigImg.src=heroProducts[0].baseImage;
    bigImg.alt=heroProducts[0].name;
    document.getElementById('solarDeviceTitle').innerText=heroProducts[0].name;
    document.getElementById('solarDeviceDesc').innerText=heroProducts[0].desc;
  }

  // Crear burbujas pequeñas
  heroProducts.forEach((prod,i)=>{
    const bubble=document.createElement('div');
    bubble.className='small-bubble';
    bubble.title=prod.name;
    const img=document.createElement('img');
    img.src=prod.baseImage;
    img.alt=prod.name;
    img.loading='lazy';
    bubble.appendChild(img);
    bubble.addEventListener('click',()=>focusBubble(bubble,prod));
    container.appendChild(bubble);
  });

  // Posicionar y animar
  bubbleRotation=0;
  positionBubbles(false);

  // Rotar cada 3 segundos
  if(bubbleIntervalId) clearInterval(bubbleIntervalId);
  bubbleIntervalId=setInterval(()=>{
    bubbleRotation=(bubbleRotation+45)%360;
    positionBubbles(true);
  },3000);
}

function positionBubbles(animate=true){
  const container=document.getElementById('bubbleSystem');
  if(!container)return;
  const bubbles=container.querySelectorAll('.small-bubble');
  if(!bubbles.length)return;
  const size=container.clientWidth;
  const radius=size*0.36;

  bubbles.forEach((bubble,i)=>{
    const angleDeg=(i*(360/bubbles.length)+bubbleRotation);
    const angleRad=angleDeg*Math.PI/180;
    const x=50+(Math.cos(angleRad)*radius/size*100);
    const y=50+(Math.sin(angleRad)*radius/size*100);
    if(!animate){
      bubble.style.transition='none';
    }
    bubble.style.left=x+'%';
    bubble.style.top=y+'%';
    if(!animate){
      // forzar reflow
      void bubble.offsetWidth;
      bubble.style.transition='';
    }
  });
}

function focusBubble(bubbleEl,prod){
  const bigBubble=document.getElementById('bigBubble');
  const bigImg=document.getElementById('bigBubbleImg');
  const title=document.getElementById('solarDeviceTitle');
  const description=document.getElementById('solarDeviceDesc');
  if(!bigBubble||!prod)return;

  // Pulso en la burbuja pequeña
  bubbleEl.classList.add('active');
  setTimeout(()=>bubbleEl.classList.remove('active'),750);

  // Flash en la burbuja grande
  bigBubble.classList.add('flash');
  setTimeout(()=>bigBubble.classList.remove('flash'),800);

  // Actualizar contenido
  setTimeout(()=>{
    bigImg.src=prod.baseImage;
    bigImg.alt=prod.name;
    title.innerText=prod.name;
    description.innerText=prod.desc;
  },180);
}

function addToCart(){
  if(!selectedProductForCart)return;
  const finalPrice=selectedProductForCart.price-(selectedProductForCart.price*(selectedProductForCart.discount/100));
  const existing=cart.find(item=>item.id===selectedProductForCart.id&&item.chosenColor===selectedColorName);
  if(existing){existing.qty+=1;}
  else{cart.push({...selectedProductForCart,chosenColor:selectedColorName,finalPrice:finalPrice,qty:1});}
  updateCartBadge();
  closeModals();
}
function getCartUnits(){return cart.reduce((sum,item)=>sum+(item.qty||1),0);}
function getCartSubtotal(){return cart.reduce((sum,item)=>sum+(item.finalPrice*(item.qty||1)),0);}
function updateCartBadge(){document.getElementById('cartCount').innerText=getCartUnits();}
function openCart(){
  const list=document.getElementById('cartItemsList');
  list.innerHTML='';
  let subtotal=getCartSubtotal();
  if(cart.length===0){list.innerHTML='<p style="text-align:center;color:var(--text-muted);padding:20px;">Tu carrito está vacío actualmente.</p>';}
  else{
    cart.forEach((item,idx)=>{
      const qty=item.qty||1;
      list.innerHTML+=`
      <div class="cart-item">
        <div style="display:flex;gap:12px;align-items:center;min-width:0;">
          <img src="${item.baseImage}" alt="${item.name}" loading="lazy">
          <div style="min-width:0;">
            <h4 style="font-size:.95rem;white-space:nowrap;overflow:hidden;text-overflow:ellipsis;">${item.name}</h4>
            <span style="font-size:.8rem;color:var(--text-muted);">Color: ${item.chosenColor}</span>
            <div class="qty-control" style="margin-top:8px;">
              <button onclick="changeCartQty(${idx},-1)" aria-label="Disminuir">−</button>
              <strong style="min-width:22px;text-align:center;">${qty}</strong>
              <button onclick="changeCartQty(${idx},1)" aria-label="Aumentar">+</button>
            </div>
          </div>
        </div>
        <div style="display:flex;gap:10px;align-items:center;">
          <strong>$${(item.finalPrice*qty).toFixed(2)}</strong>

  document.getElementById('cartDiscount').innerText='-$'+discount.toFixed(2);
  document.getElementById('cartTotal').innerText='$'+total.toFixed(2);
  document.getElementById('modalCart').classList.add('active');
}
function changeCartQty(index,delta){
  if(!cart[index])return;
  cart[index].qty=(cart[index].qty||1)+delta;
  if(cart[index].qty<=0)cart.splice(index,1);
  updateCartBadge();openCart();
}
function removeFromCart(index){cart.splice(index,1);updateCartBadge();openCart();}

function processCheckout(payMethod){
  if(!requireAuth('login'))return;
  if(cart.length===0)return;
  generateFormalPDF(payMethod);
  const date=new Date().toLocaleString('es-SV');
  const subtotal=getCartSubtotal();
  const discount=subtotal>500?subtotal*0.05:0;
  const total=subtotal-discount;
  const order={
    id:'NX-'+Date.now().toString().slice(-8),date,status:'Procesada',payMethod,total,

}

  const rows=cart.map(item=>[String(item.qty||1),item.name,item.chosenColor,`${item.specs.ram} RAM | ${item.specs.storage}`,"$"+item.finalPrice.toFixed(2)]);
  doc.autoTable({startY:75,head:[columns],body:rows,theme:'grid',headStyles:{fillColor:[11,17,32],textColor:[255,255,255]}});
  const finalY=doc.lastAutoTable.finalY||100;
  let subtotal=cart.reduce((acc,curr)=>acc+(curr.finalPrice*(curr.qty||1)),0);
  let tax=subtotal*0.13;
  let total=subtotal;
  doc.setFont("helvetica","bold");
  doc.text(`Subtotal Exento/Gravado: $${(subtotal-tax).toFixed(2)}`,130,finalY+15);
  doc.text(`IVA Aplicado (13%): $${tax.toFixed(2)}`,130,finalY+22);
  doc.setFontSize(12);doc.setTextColor(16,185,129);
  doc.text(`TOTAL FINAL PAGADO: $${total.toFixed(2)}`,130,finalY+30);
  doc.setTextColor(100,100,100);doc.setFontSize(8);
  doc.text("Este comprobante constituye un título ejecutivo de garantía oficial por 12 meses en cualquier sucursal Nexus GX de El Salvador.",15,270);
  doc.text("Nexus GX El Salvador - Inframen / CelularSync Pro © 2026",15,275);

}

async function handleRegister(){
  const firstName=document.getElementById('regFirstName').value.trim();
  const lastName=document.getElementById('regLastName').value.trim();
  const username=normalizeUsername(document.getElementById('regUsername').value);
  const email=normalizeUsername(document.getElementById('regEmail').value);
  const password=document.getElementById('regPassword').value;
  const confirmPassword=document.getElementById('regConfirmPassword').value;
  if(!firstName||!lastName||!username||!email||!password||!confirmPassword){setFormFeedback('registerFeedback','Completa todos los campos para crear tu cuenta.','error');return;}
  if(!/^[a-zA-ZÀ-ÿ0-9_]{4,20}$/.test(username)){setFormFeedback('registerFeedback','El usuario debe tener entre 4 y 20 caracteres y solo usar letras, números o guion bajo.','error');return;}
  if(!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)){setFormFeedback('registerFeedback','Ingresa un correo electrónico válido.','error');return;}
  if(!/(?=.*[a-z])(?=.*[A-Z])(?=.*\d).{8,}/.test(password)){setFormFeedback('registerFeedback','La contraseña debe tener al menos 8 caracteres, una mayúscula, una minúscula y un número.','error');return;}
  if(password!==confirmPassword){setFormFeedback('registerFeedback','La confirmación de contraseña no coincide.','error');return;}
  const users=getUsers();
  if(users.some(u=>u.username===username)){setFormFeedback('registerFeedback','Ese usuario ya está registrado. Prueba con otro.','error');return;}
  if(users.some(u=>u.email===email)){setFormFeedback('registerFeedback','Ese correo ya está registrado. Puedes iniciar sesión con él.','error');return;}
  const passwordHash=await hashPassword(password);
  const user={id:(window.crypto&&typeof window.crypto.randomUUID==='function')?window.crypto.randomUUID():`u-${Date.now()}`,firstName,lastName,username,email,passwordHash,createdAt:new Date().toISOString()};
  users.push(user);saveUsers(users);
  currentUser=user;
  sessionStorage.setItem(STORAGE_KEYS.session,JSON.stringify(user));
  setFormFeedback('registerFeedback','Cuenta creada correctamente. Tu sesión se inició automáticamente.','success');
  updateAuthButton();updateProfileUI();renderPurchaseHistory();
  setTimeout(()=>{closeModals();navigate('profile');},700);
}
async function handleLogin(){
  const identifier=normalizeUsername(document.getElementById('logUser').value);
  const password=document.getElementById('logPass').value;
  if(!identifier||!password){setFormFeedback('loginFeedback','Escribe tu usuario/correo y contraseña.','error');return;}
  const users=getUsers();
  const user=users.find(item=>item.username===identifier||item.email===identifier);
  if(!user){setFormFeedback('loginFeedback','No encontramos una cuenta con esos datos. Crea una cuenta para continuar.','error');return;}
  const passwordHash=await hashPassword(password);
  if(passwordHash!==user.passwordHash){setFormFeedback('loginFeedback','La contraseña es incorrecta. Revisa tus datos e inténtalo nuevamente.','error');return;}
  currentUser=user;
  sessionStorage.setItem(STORAGE_KEYS.session,JSON.stringify(user));
  updateAuthButton();updateProfileUI();renderPurchaseHistory();
  setFormFeedback('loginFeedback',`¡Bienvenido, ${user.firstName}! Inicio de sesión exitoso.`,'success');
  setTimeout(()=>{closeModals();},550);
}

}

function filterByCategory(cat){

function filterProducts(){
  const query=document.getElementById('searchInput').value.toLowerCase().trim();
  const filtered=products.filter(p=>
    p.name.toLowerCase().includes(query)||
    (p.brand||'').toLowerCase().includes(query)||
    p.category.toLowerCase().includes(query)||
    p.desc.toLowerCase().includes(query)||
    p.specs.ram.toLowerCase().includes(query)||
    p.specs.storage.toLowerCase().includes(query)||
    p.specs.processor.toLowerCase().includes(query)||
    p.specs.wifi.toLowerCase().includes(query)
  );
  renderProducts(filtered);
}
